const router = require("express").Router(); const db = require("../config/database"); const auth = require("../middleware/auth"); // ============================== // 获取护航套餐列表 // ============================== router.get("/packages", async (req,res)=>{ const [rows] = await db.query( ` SELECT * FROM packages WHERE status=1 ORDER BY sort ASC ` ); res.json({ success:true, data:rows }); }); // ============================== // 创建订单 // ============================== router.post( "/create", auth, async(req,res)=>{ const { package_id }=req.body; //查询套餐 const [pkg]=await db.query( "SELECT * FROM packages WHERE id=?", [package_id] ); if(!pkg.length){ return res.json({ message:"套餐不存在" }); } let price=pkg[0].price; //查询用户VIP const [user]=await db.query( "SELECT vip_level FROM users WHERE id=?", [req.user.id] ); let discount=1; if(user[0].vip_level==="银牌"){ discount=0.98; } if(user[0].vip_level==="金牌"){ discount=0.95; } if(user[0].vip_level==="黑金"){ discount=0.90; } let finalPrice= Number(price)*discount; //生成订单号 let orderNo= "LMP"+Date.now(); await db.query( ` INSERT INTO orders ( order_no, user_id, package_id, original_price, discount_price, status, insurance_target ) VALUES(?,?,?,?,?,?,?) `, [ orderNo, req.user.id, package_id, price, finalPrice, "待接单", pkg[0].insurance_amount ] ); res.json({ success:true, order_no:orderNo, pay_price:finalPrice }); }); // ============================== // 用户订单列表 // ============================== router.get( "/my", auth, async(req,res)=>{ const [rows]=await db.query( ` SELECT o.*, p.package_name FROM orders o LEFT JOIN packages p ON o.package_id=p.id WHERE o.user_id=? ORDER BY o.id DESC `, [req.user.id] ); res.json({ success:true, data:rows }); }); // ============================== // 管理员查看全部订单 // ============================== router.get( "/all", auth, async(req,res)=>{ const [rows]=await db.query( ` SELECT o.*, u.username, p.package_name FROM orders o LEFT JOIN users u ON o.user_id=u.id LEFT JOIN packages p ON o.package_id=p.id ORDER BY o.id DESC ` ); res.json({ success:true, data:rows }); }); // ============================== // 更新保单进度 // ============================== router.post( "/progress", auth, async(req,res)=>{ const { order_id, progress, insurance_current }=req.body; let status="进行中"; if(progress>=100){ status="已完成"; } await db.query( ` UPDATE orders SET progress=?, insurance_current=?, status=? WHERE id=? `, [ progress, insurance_current, status, order_id ] ); res.json({ success:true }); }); // ============================== // 失败自动续打 // ============================== router.post( "/failed", auth, async(req,res)=>{ const { order_id }=req.body; await db.query( ` UPDATE orders SET status='失败续打' WHERE id=? `, [order_id] ); res.json({ success:true, message:"已进入免费续打流程" }); }); module.exports=router;
